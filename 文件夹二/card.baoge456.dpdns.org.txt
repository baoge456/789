// Function to update an existing link
async function updateLink(originalLink) {
console.log('DEBUG: updateLink called for original link:', originalLink);
if (!await validateToken()) {
return;
}

const nameInput = document.getElementById('name-input');
const urlInput = document.getElementById('url-input');
const tipsInput = document.getElementById('tips-input');
const iconInput = document.getElementById('icon-input');
const categorySelect = document.getElementById('category-select');
const privateStatusSelect = document.getElementById('private-status-select'); // Changed from checkbox

const newName = nameInput ? nameInput.value.trim() : '';
const newUrl = urlInput ? urlInput.value.trim() : '';
const newTips = tipsInput ? tipsInput.value.trim() : '';
const newIcon = iconInput ? iconInput.value.trim() : '';
const newCategory = categorySelect ? categorySelect.value : '';
const newIsPrivate = privateStatusSelect ? (privateStatusSelect.value === 'private') : false; // Get from select
console.log('DEBUG: updateLink - newIsPrivate from select:', newIsPrivate); // 调试日志

// Validation
if (!newName || !newUrl || !newCategory) {
await customAlert('名称、URL和分类是必填项', '编辑链接');
return;
}
if (!isValidHttpUrl(newUrl)) {
await customAlert('URL必须以http://或https://开头', '编辑链接');
return;
}
if (newIcon && !isValidHttpUrl(newIcon)) {
await customAlert('图标URL必须以http://或https://开头，或留空', '编辑链接');
return;
}

// Check for duplicate URL if URL changed
if (newUrl.toLowerCase() !== originalLink.url.toLowerCase()) {
const allLinks = [...publicLinks, ...privateLinks];
const isUrlExists = allLinks.some(link => link.url.toLowerCase() === newUrl.toLowerCase());
if (isUrlExists) {
await customAlert('该URL已存在，请勿重复添加', '编辑链接');
return;
}
}

// Remove old link from its category and global lists
let oldLinkArray = originalLink.isPrivate ? privateLinks : publicLinks;
let oldLinkIndex = oldLinkArray.findIndex(link => link.url === originalLink.url);
if (oldLinkIndex !== -1) {
oldLinkArray.splice(oldLinkIndex, 1);
}

// Remove from categories object (important for category structure)
if (categories[originalLink.category] && categories[originalLink.category].links) {
categories[originalLink.category].links = categories[originalLink.category].links.filter(link => link.url !== originalLink.url);
}

// Update link properties
originalLink.name = newName;
originalLink.url = newUrl;
originalLink.tips = newTips;
originalLink.icon = newIcon;
originalLink.category = newCategory;
originalLink.isPrivate = newIsPrivate; // <--- 这里更新了 originalLink 的 isPrivate
console.log('DEBUG: updateLink - originalLink.isPrivate after update:', originalLink.isPrivate); // 调试日志

// Add updated link to new category and global lists
if (newIsPrivate) {
privateLinks.push(originalLink);
} else {
publicLinks.push(originalLink);
}

// Add to categories object (important for category structure)
if (!categories[newCategory]) {
// 如果新分类不存在，默认创建为公开分类
categories[newCategory] = { isPrivate: false, links: [] };
}
categories[newCategory].links.push(originalLink); // <--- 这里推入的是 originalLink 的引用

links = isLoggedIn ? [...publicLinks, ...privateLinks] : publicLinks;

hideAddDialog();
renderSections(); // Re-render to reflect changes
saveLinks();
}

async function addCategory() {
if (!isAdmin) {
customAlert('请先进入设置模式才能添加分类。', '添加分类');
return;
}
if (!await validateToken()) {
return;
}
console.log('DEBUG: addCategory called.');
const result = await showCategoryDialog('请输入新分类名称'); // result is { name, isPrivate }
if (result && result.name) {
const categoryName = result.name;
const isPrivate = result.isPrivate; // <-- 这里获取了 isPrivate
console.log('DEBUG: addCategory - new category name:', categoryName, 'isPrivate from dialog:', isPrivate); // 调试日志
if (categories[categoryName]) {
await customAlert('该分类已存在', '添加分类');
} else {
categories[categoryName] = { isPrivate: isPrivate, links: [] }; // <-- 这里设置了分类的 isPrivate
console.log('DEBUG: addCategory - categories[' + categoryName + '].isPrivate after creation:', categories[categoryName].isPrivate); // 调试日志
updateCategorySelect();
renderSections(); // Re-render to show new category
saveLinks();
}
} else {
console.log('DEBUG: addCategory - Dialog cancelled or no name provided.');
}
}

// 删除分类
async function deleteCategory(categoryName) {
if (!isAdmin) {
customAlert('请先进入设置模式才能删除分类。', '删除分类');
return;
}
if (!await validateToken()) {
return;
}
console.log('DEBUG: deleteCategory called for:', categoryName);
const message = '确定要删除 "' + categoryName + '" 分类吗？这将删除该分类下的所有链接。';
const confirmed = await customConfirm(message, '确定', '取消');

if (confirmed) {
// 从全局 links, publicLinks, privateLinks 中过滤掉属于该分类的所有链接
links = links.filter(link => link.category !== categoryName);
publicLinks = publicLinks.filter(link => link.category !== categoryName);
privateLinks = privateLinks.filter(link => link.category !== categoryName);

// 删除整个分类对象
delete categories[categoryName];
console.log('DEBUG: deleteCategory - Category', categoryName, 'deleted from categories object.');

updateCategorySelect();
renderSections();
renderCategoryButtons();
saveLinks();
} else {
console.log('DEBUG: deleteCategory - Deletion cancelled by user.');
}
}

// 编辑分类名称
async function editCategoryName(oldName) {
if (!isAdmin) {
customAlert('请先进入设置模式才能编辑分类。', '编辑分类');
return;
}
if (!await validateToken()) return;

const oldCategoryData = categories[oldName];
if (!oldCategoryData) {
await customAlert('分类不存在。', '编辑分类');
return;
}
console.log('DEBUG: editCategoryName - called for:', oldName, 'oldCategoryData.isPrivate before dialog:', oldCategoryData.isPrivate); // 调试日志

// 传入当前分类的 isPrivate 状态
const result = await showCategoryDialog('请输入新的分类名称', oldName, oldCategoryData.isPrivate);
if (!result || !result.name) { // 允许名称不变，只改私密状态
console.log('DEBUG: editCategoryName - Dialog cancelled or no name provided.');
return;
}

const newName = result.name;
const newIsPrivate = result.isPrivate; // <-- 这里获取了新的 isPrivate
console.log('DEBUG: editCategoryName - result.name from dialog:', newName, 'result.isPrivate from dialog:', newIsPrivate); // 调试日志

if (newName !== oldName && categories[newName]) { // 只有在名称改变时才检查重复
await customAlert('该名称已存在，请重新命名', '编辑分类');
return;
}

// 1. 更新 oldCategoryData 的 isPrivate 属性
oldCategoryData.isPrivate = newIsPrivate; // <-- 这里更新了 oldCategoryData 的 isPrivate
console.log('DEBUG: editCategoryName - oldCategoryData.isPrivate after update from dialog:', oldCategoryData.isPrivate); // 调试日志

// 2. 如果名称改变，则重命名分类对象
if (newName !== oldName) {
categories[newName] = oldCategoryData; // <-- 引用赋值，oldCategoryData 已经是 categories[newName]
delete categories[oldName];
console.log('DEBUG: editCategoryName - Category name changed from', oldName, 'to', newName); // 调试日志

// 3. 更新所有链接的 category 字段
[...publicLinks, ...privateLinks].forEach(link => {
if (link.category === oldName) {
link.category = newName;
}
});

links.forEach(link => {
if (link.category === oldName) {
link.category = newName;
}
});
} else {
console.log('DEBUG: editCategoryName - Category name did not change.'); // 调试日志
}

console.log('DEBUG: editCategoryName - Final categories object after edit:', JSON.parse(JSON.stringify(categories))); // 调试日志

// 4. 保存并刷新
renderSections();
renderCategoryButtons();
updateCategorySelect();
saveLinks();
}

// 移动分类
async function moveCategory(categoryName, direction) {
console.log('DEBUG: moveCategory called for:', categoryName, 'direction:', direction);
if (!isAdmin) {
customAlert('请先进入设置模式才能移动分类。', '移动分类');
return;
}
if (!await validateToken()) {
console.log('DEBUG: validateToken failed in moveCategory.');
return;
}

const keys = Object.keys(categories);
const index = keys.indexOf(categoryName);
console.log('DEBUG: Current category keys (before move):', keys.join(', ')); // Existing log
console.log('DEBUG: Index of', categoryName, 'is', index); // Existing log

if (index < 0) {
console.warn('DEBUG: Category not found in keys:', categoryName);
return;
}

const newIndex = index + direction;
if (newIndex < 0 || newIndex >= keys.length) {
console.log('DEBUG: New index out of bounds:', newIndex);
return;
}

// Create a new array for reordered keys
const reorderedKeys = [...keys];
// Swap elements in the new array
[reorderedKeys[index], reorderedKeys[newIndex]] = [reorderedKeys[newIndex], reorderedKeys[index]];
console.log('DEBUG: Reordered keys array:', reorderedKeys.join(', ')); // Existing log

// Reconstruct the categories object to preserve the new order
const newCategories = {};
reorderedKeys.forEach(key => {
newCategories[key] = categories[key];
});
console.log('DEBUG: newCategories object (reconstructed order):', Object.keys(newCategories).join(', ')); // Existing log

// Clear the original categories object
Object.keys(categories).forEach(k => delete categories[k]);
// Assign the reordered categories back to the original object
Object.assign(categories, newCategories);
console.log('DEBUG: categories object after reordering (final state):', Object.keys(categories).join(', ')); // Existing log

// Re-render UI elements to reflect the new order
renderSections();
renderCategoryButtons();
updateCategorySelect();
saveLinks();
console.log('DEBUG: moveCategory finished, UI re-rendered and links saved.'); // Existing log
}

// 加载特定分类的卡片
function loadCategoryCards(categoryName) {
const categoryData = categories[categoryName];
if (!categoryData) {
console.warn('DEBUG: loadCategoryCards - Category data not found for:', categoryName);
return;
}

console.log('DEBUG: loadCategoryCards - Category:', categoryName, 'categoryData.isPrivate:', categoryData.isPrivate, 'isAdmin:', isAdmin, 'isLoggedIn:', isLoggedIn); // 调试日志

// 【核心逻辑】如果分类是私密的且用户无权查看，则直接返回
if (categoryData.isPrivate && !isLoggedIn) { // isAdmin 可以看到所有分类，所以这里只检查 isLoggedIn
console.warn('DEBUG: Skipping loading private category for non-logged-in user:', categoryName);
return;
}

// 如果已经是当前激活分类，则收缩该分类
if (activeCategory === categoryName) {
// 收缩当前分类
const currentSection = document.querySelector('.section[data-category="' + categoryName + '"]');
if (currentSection) {
currentSection.style.display = 'none';
console.log('DEBUG: loadCategoryCards - Collapsing active category:', categoryName);
}

// 重置激活分类
activeCategory = null;

// 更新分类按钮的激活状态
updateActiveCategoryButton(null);

return;
}

// 非管理员模式下，隐藏所有分类区域，只显示当前选中的
if (!isAdmin) {
document.querySelectorAll('.section').forEach(section => {
section.style.display = 'none';
});

const currentSection = document.querySelector('.section[data-category="' + categoryName + '"]');
if (currentSection) {
currentSection.style.display = 'block';
console.log('DEBUG: loadCategoryCards - Displaying selected category (non-admin):', categoryName);
}
} else {
// 管理员模式下显示所有分类
document.querySelectorAll('.section').forEach(section => {
section.style.display = 'block';
});
console.log('DEBUG: loadCategoryCards - Displaying all categories (admin mode).');
}

const cardContainer = document.getElementById(categoryName);
if (!cardContainer) {
console.error('ERROR: loadCategoryCards - Card container not found for category:', categoryName);
return;
}

// 检查是否已经加载过
if (cardContainer.getAttribute('data-loaded') === 'true') {
// 已加载，只需要滚动到该分类
scrollToCategory(categoryName);
activeCategory = categoryName;
updateActiveCategoryButton(categoryName);
console.log('DEBUG: loadCategoryCards - Category already loaded, scrolling to:', categoryName);
return;
}

// 清空容器
cardContainer.innerHTML = '';

// 加载该分类的卡片
(categoryData.links || []).forEach(link => {
// 【核心逻辑】如果分类是私密的，其下所有链接都视为私密
// 如果分类是公开的，则根据链接自身的 isPrivate 属性判断
const isLinkVisible = !categoryData.isPrivate || isLoggedIn; // 只有当分类是公开的 OR 用户已登录，链接才可能可见

if (isLinkVisible) {
createCard(link, cardContainer);
} else {
console.log('DEBUG: loadCategoryCards - Skipping link (due to category privacy or link privacy) for non-logged-in user:', link.name);
}
});

// 标记为已加载
cardContainer.setAttribute('data-loaded', 'true');

// 滚动到该分类
scrollToCategory(categoryName);

// 更新当前激活分类
activeCategory = categoryName;

// 更新分类按钮的激活状态
updateActiveCategoryButton(categoryName);
console.log('DEBUG: loadCategoryCards - Finished loading and displaying category:', categoryName);
}

if (isShowingSearchResults) {

buttonsContainer.innerHTML = '';

// 只有当有分类时才显示按钮容器

// 根据可见性设置活跃的分类按钮
function setActiveCategoryButtonByVisibility() {
console.log('DEBUG: setActiveCategoryButtonByVisibility called.');
// 如果正在显示搜索结果，不更新分类按钮的活跃状态
if (isShowingSearchResults) {
console.log('DEBUG: setActiveCategoryButtonByVisibility skipped because search results are showing.');
return;
}

// 获取所有分类区域
const sections = document.querySelectorAll('.section');
if (!sections.length) {
console.log('DEBUG: setActiveCategoryButtonByVisibility - No sections found.');
return;
}

// 获取视窗高度
const viewportHeight = window.innerHeight;
// 考虑固定元素的高度
const fixedElementsHeight = 170;
// 计算视窗中心点
const viewportCenter = viewportHeight / 2 + fixedElementsHeight;

// 找出最接近视窗中心的分类
let closestSection = null;
let closestDistance = Infinity;

sections.forEach(section => {
// 只有可见的 section 才参与计算
if (section.style.display === 'none') {
console.log('DEBUG: setActiveCategoryButtonByVisibility - Skipping hidden section:', section.dataset.category);
return;
}

const rect = section.getBoundingClientRect();
// 计算分类区域的中心点
const sectionCenter = rect.top + rect.height / 2;
// 计算到视窗中心的距离
const distance = Math.abs(sectionCenter - viewportCenter);

if (distance < closestDistance) {
closestDistance = distance;
closestSection = section;
}
});

if (closestSection) {
const categoryId = closestSection.dataset.category;
console.log('DEBUG: setActiveCategoryButtonByVisibility - Closest section:', categoryId);
const buttons = document.querySelectorAll('.category-button');

// 移除所有活跃状态
buttons.forEach(btn => btn.classList.remove('active'));

// 为匹配的分类按钮添加活跃状态
buttons.forEach(btn => {
if (btn.dataset.category === categoryId) {
btn.classList.add('active');
console.log('DEBUG: setActiveCategoryButtonByVisibility - Setting active class for button:', categoryId);
}
});
} else {
console.log('DEBUG: setActiveCategoryButtonByVisibility - No closest section found.');
}
}

// 添加滚动事件监听器，滚动时更新活跃的分类按钮
window.addEventListener('scroll', debounce(setActiveCategoryButtonByVisibility, 100));

// 防抖函数，避免过多的滚动事件处理
function debounce(func, wait) {
let timeout;
return function() {
const context = this;
const args = arguments;
clearTimeout(timeout);
timeout = setTimeout(() => {
func.apply(context, args);
}, wait);
};
}

// 滚动到指定分类
function scrollToCategory(category) {
const section = document.querySelector('.section[data-category="' + category + '"]');
if (section) {
// 计算滚动位置，考虑顶部固定元素的高度和额外偏移量
let offset = 230; // 减小偏移量，确保分类标题和第一行书签完全可见

// 检查是否为移动设备
if (window.innerWidth <= 480) {
offset = 120; // 移动设备上的偏移量
}

// 滚动到分类位置
const sectionRect = section.getBoundingClientRect();
const absoluteTop = window.pageYOffset + sectionRect.top - offset;

// 使用平滑滚动效果
window.scrollTo({
top: absoluteTop,
behavior: 'smooth'
});
console.log('DEBUG: scrollToCategory - Scrolled to category:', category);
} else {
console.warn('DEBUG: scrollToCategory - Section not found for category:', category);
}
}

// 读取链接数据
async function loadLinks() {
console.log('DEBUG: loadLinks function started.'); // DEBUG
const headers = {
'Content-Type': 'application/json'
};

// 如果已登录，从 localStorage 获取 token 并添加到请求头
if (isLoggedIn) {
const token = localStorage.getItem('authToken');
if (token) {
headers['Authorization'] = token;
console.log('DEBUG: Authorization token added to headers.'); // DEBUG
} else {
console.log('DEBUG: isLoggedIn is true but no authToken found in localStorage.'); // DEBUG
}
} else {
console.log('DEBUG: User is not logged in. Fetching public links only.'); // DEBUG
}

try {
console.log('DEBUG: Fetching links from /api/getLinks?userId=testUser'); // DEBUG
const response = await fetch('/api/getLinks?userId=testUser', {
headers: headers
});
console.log('DEBUG: Response received for /api/getLinks. Status:', response.status); // DEBUG

if (!response.ok) {
const errorText = await response.text(); // Get more details about the error
console.error('ERROR: /api/getLinks HTTP error! Status:', response.status, 'Response text:', errorText); // DEBUG
throw new Error("HTTP error! status: " + response.status + ", response: " + errorText);
}

const data = await response.json();
console.log('DEBUG: Received data from /api/getLinks:', JSON.parse(JSON.stringify(data))); // DEBUG - 打印完整数据

if (data.categories) {
const tempCategories = {};
for (const catName in data.categories) {
const catContent = data.categories[catName];
if (Array.isArray(catContent)) { // 旧格式: { "Cat": [link1, link2] }
tempCategories[catName] = { isPrivate: false, links: catContent }; // 默认设为公开
console.log('DEBUG: loadLinks - Converted old category format for:', catName, 'isPrivate set to false.');
} else if (catContent && typeof catContent === 'object' && catContent.links) { // 新格式: { "Cat": { isPrivate: bool, links: [...] } }
// 确保 isPrivate 是布尔值
tempCategories[catName] = {
isPrivate: !!catContent.isPrivate,
links: catContent.links
};
console.log('DEBUG: loadLinks - Processed category:', catName, 'isPrivate:', tempCategories[catName].isPrivate);
} else {
console.warn('DEBUG: loadLinks - Unexpected category format for', catName, ':', catContent, '. Initializing as public and empty.');
tempCategories[catName] = { isPrivate: false, links: [] }; // 兜底处理
}
}
categories = tempCategories;
console.log('DEBUG: loadLinks - Categories object keys received from KV and assigned:', Object.keys(categories).join(', ')); // Added debug log
console.log('DEBUG: loadLinks - Categories object AFTER processing data.categories (final state):', JSON.parse(JSON.stringify(categories))); // 调试日志
} else {
console.warn('DEBUG: No categories received from API. Initializing categories as empty object.'); // DEBUG
categories = {}; // Ensure it's an empty object if not provided
}

// Flatten links from categories for publicLinks, privateLinks, and links arrays
publicLinks = [];
privateLinks = [];
links = []; // This will contain all links if logged in, or only public links if not.

Object.keys(categories).forEach(catName => {
const categoryData = categories[catName];
if (categoryData && Array.isArray(categoryData.links)) {
categoryData.links.forEach(link => {
// Ensure link.isPrivate is boolean
link.isPrivate = !!link.isPrivate;
if (link.isPrivate) {
privateLinks.push(link);
} else {
publicLinks.push(link);
}
// If logged in, add all links to 'links' array. Otherwise, only public links.
// 【核心逻辑】这里不应该根据 categoryData.isPrivate 过滤，因为 links 数组是所有链接的扁平化列表，
// 它的可见性判断应该在渲染时进行。
links.push(link);
});
}
});

console.log('DEBUG: Global links array updated. Total links:', links.length); // DEBUG
console.log('DEBUG: Public links:', publicLinks.length, 'Private links:', privateLinks.length); // DEBUG

} catch (error) {
console.error('ERROR: 加载链接失败:', error); // DEBUG
customAlert('无法加载数据，请检查网络或联系管理员。错误: ' + error.message, '加载失败');
// Ensure categories is an empty object on error to prevent further issues
categories = {};
links = []; // Also clear links on error
publicLinks = [];
privateLinks = [];
}
console.log('DEBUG: loadLinks function finished.'); // DEBUG
}

// 从URL中提取域名
function extractDomain(url) {
let domain;
try {
const parsedUrl = new URL(url);
domain = parsedUrl.hostname;
} catch (e) {
// 如果 URL 无效，直接返回原始 URL
domain = url;
}
return domain;
}

// URL验证函数
function isValidUrl(url) {
try {
new URL(url);
return true;
} catch {
return false;
}
}
// URL协议验证函数 (HTTP/HTTPS only) - 前端新增
function isValidHttpUrl(string) {
let url;
try {
url = new URL(string);
} catch (_) {
return false;
}
return url.protocol === "http:" || url.protocol === "https:";
}

// 保存链接数据
// 防抖变量
let saveLinksTimeoutId = null;
let pendingSaveLinks = false;

// 优化后的保存链接数据函数（带防抖）
async function saveLinks(debounceMs = 1000) {
console.log('DEBUG: saveLinks called.');

// 如果已有防抖定时器，清除它并重新设置
if (saveLinksTimeoutId) {
clearTimeout(saveLinksTimeoutId);
}

// 设置新的防抖定时器
saveLinksTimeoutId = setTimeout(async () => {
try {
if (isAdmin && !(await validateToken())) {
console.log('DEBUG: saveLinks - Token validation failed, aborting save.');
return;
}

let allLinks = [...publicLinks, ...privateLinks];

// 保留链接的实际状态，不强制修改
// 如果链接没有状态信息，设置为默认值
allLinks.forEach(link => {
if (!link.status) {
link.status = 'ok'; // 仅为没有状态的链接设置默认值
}
if (!link.lastChecked) {
link.lastChecked = new Date().toISOString();
}
// 确保 isPrivate 是布尔值
if (typeof link.isPrivate !== 'boolean') {
link.isPrivate = !!link.isPrivate; // Convert to boolean
}
console.log('DEBUG: saveLinks - Link:', link.name, 'isPrivate:', link.isPrivate);
});

// 确保 categories 对象中的 isPrivate 属性是布尔值
for (const catName in categories) {
if (categories[catName] && typeof categories[catName].isPrivate !== 'boolean') {
categories[catName].isPrivate = !!categories[catName].isPrivate;
}
console.log('DEBUG: saveLinks - Category:', catName, 'isPrivate:', categories[catName].isPrivate);
}
console.log('DEBUG: saveLinks - Categories object keys being sent to KV:', Object.keys(categories).join(', ')); // Added debug log
console.log('DEBUG: saveLinks - Categories object BEFORE JSON.stringify and API call:', JSON.parse(JSON.stringify(categories))); // 调试日志

const response = await fetch('/api/saveOrder', {
method: 'POST',
headers: {
'Content-Type': 'application/json',
'Authorization': localStorage.getItem('authToken')
},
body: JSON.stringify({
userId: 'testUser',
links: allLinks,
categories: categories
}),
});

if (!response.ok) {
const errorText = await response.text();
throw new Error('服务器响应错误: ' + response.status + ', response: ' + errorText);
}

const result = await response.json();
if (!result.success) {
throw new Error('保存失败: ' + (result.message || '未知错误'));
}

console.log('DEBUG: 保存链接成功，包含状态信息. Result:', result);
showToast('保存成功！');
} catch (error) {
console.error('ERROR: 保存链接失败，请重试:', error);
showToast('保存链接失败: ' + (error.message || '请重试'));
}
console.log('DEBUG: saveLinks finished.');

// 重置防抖标志
pendingSaveLinks = false;
}, debounceMs);

// 设置防抖标志
pendingSaveLinks = true;
}

// 添加卡片
async function addLink() {
console.log('DEBUG: addLink called.');
if (!await validateToken()) {
console.log('DEBUG: addLink - Token validation failed, aborting add.');
return;
}
const nameInput = document.getElementById('name-input');
const urlInput = document.getElementById('url-input');
const tipsInput = document.getElementById('tips-input');
const iconInput = document.getElementById('icon-input');
const categorySelect = document.getElementById('category-select');
const privateStatusSelect = document.getElementById('private-status-select'); // Changed from checkbox

const name = nameInput ? nameInput.value.trim() : '';
const url = urlInput ? urlInput.value.trim() : '';
const tips = tipsInput ? tipsInput.value.trim() : '';
const icon = iconInput ? iconInput.value.trim() : '';
const category = categorySelect ? categorySelect.value : '';
const isPrivate = privateStatusSelect ? (privateStatusSelect.value === 'private') : false; // Get from select
console.log('DEBUG: addLink - new link isPrivate from select:', isPrivate); // 调试日志

// 验证必填字段
if (!name || !url || !category) {
let errorMessage = '';
if (!name && !url) {
errorMessage = '请输入名称和URL';
} else if (!name) {
errorMessage = '请输入名称';
} else if (!url) {
errorMessage = '请输入URL';
}

await customAlert(errorMessage, '添加卡片');
if (!name && nameInput) {
nameInput.focus();
} else if (!url && urlInput) {
urlInput.focus();
}
return;
}
// URL协议验证
if (!isValidHttpUrl(url)) {
await customAlert('URL必须以http://或https://开头', '添加卡片');
if (urlInput) urlInput.focus();
return;
}
if (icon && !isValidHttpUrl(icon)) {
await customAlert('图标URL必须以http://或https://开头，或留空', '添加卡片');
if (iconInput) iconInput.focus();
return;
}

// 检查URL是否已存在
const normalizedUrl = url.toLowerCase();
const allLinks = [...publicLinks, ...privateLinks];
const isUrlExists = allLinks.some(link => link.url.toLowerCase() === normalizedUrl);

if (isUrlExists) {
await customAlert('该URL已存在，请勿重复添加', '添加卡片');
if (urlInput) urlInput.focus();
return;
}

const newLink = { name, url, tips, icon, category, isPrivate };
console.log('DEBUG: addLink - newLink object:', newLink); // 调试日志

if (isPrivate) {
privateLinks.push(newLink);
} else {
publicLinks.push(newLink);
}

// Add to categories object
if (!categories[category]) {
// 如果是新分类，默认创建为公开分类
categories[category] = { isPrivate: false, links: [] };
console.log('DEBUG: addLink - Created new category:', category, 'as public.');
}
categories[category].links.push(newLink);
console.log('DEBUG: addLink - Added link to category:', category);

links = isLoggedIn ? [...publicLinks, ...privateLinks] : publicLinks;
console.log('DEBUG: addLink - Global links array updated. Total links:', links.length);

// 如果分类本身是私密的，那么只有登录用户才能看到其链接
const categoryData = categories[category];
const isCategoryVisible = !categoryData.isPrivate || isLoggedIn;

if (isCategoryVisible) {
const container = document.getElementById(category);
if (container) {
// If the category is currently loaded, add the card directly
if (container.getAttribute('data-loaded') === 'true') {
// 只有当链接本身可见时才创建卡片
const isLinkVisibleInContext = !categoryData.isPrivate || isLoggedIn; // Redundant check, but for clarity
if (isLinkVisibleInContext) {
createCard(newLink, container);
console.log('DEBUG: addLink - Created card directly in loaded container.');
} else {
console.log('DEBUG: addLink - Link not visible in context, skipping direct card creation.');
}
} else {
// If not loaded, re-render sections to ensure it appears when category is expanded
renderSections();
console.log('DEBUG: addLink - Re-rendered sections because container not loaded.');
}
} else {
// If category container doesn't exist, it means it's a new category or not rendered yet
renderSections();
console.log('DEBUG: addLink - Re-rendered sections because category container not found.');
}
} else {
console.log('DEBUG: addLink - Category is not visible, skipping card creation/section re-render.');
}

saveLinks();

// 清空表单
if (nameInput) nameInput.value = '';
if (urlInput) urlInput.value = '';
if (tipsInput) tipsInput.value = '';
if (iconInput) iconInput.value = '';
if (privateStatusSelect) privateStatusSelect.value = 'public'; // Reset to public
hideAddDialog();
console.log('DEBUG: addLink finished.');
}

if (!await validateToken()) {

// 拖拽卡片
let draggedCard = null;
let touchStartX, touchStartY;

// 触屏端拖拽卡片
function touchStart(event) {
// ★★ 解决按钮点击无效 ★★
if (event.target.closest('.card-btn')) {
return; // 点到按钮就不进入拖拽流程
}

if (!isAdmin) return;

draggedCard = event.target.closest('.card');
if (!draggedCard) return;

event.preventDefault();
const touch = event.touches[0];
touchStartX = touch.clientX;
touchStartY = touch.clientY;

draggedCard.classList.add('dragging');

document.addEventListener('touchmove', touchMove, { passive: false });
document.addEventListener('touchend', touchEnd);
}

function touchMove(event) {
if (!draggedCard) return;
event.preventDefault();

const touch = event.touches[0];
const currentX = touch.clientX;
const currentY = touch.clientY;

const deltaX = currentX - touchStartX;
const deltaY = currentY - touchStartY;
draggedCard.style.transform = "translate(" + deltaX + "px, " + deltaY + "px)";

const target = findCardUnderTouch(currentX, currentY);
if (target && target !== draggedCard) {
const container = target.parentElement;
const targetRect = target.getBoundingClientRect();

if (currentX < targetRect.left + targetRect.width / 2) {
container.insertBefore(draggedCard, target);
} else {
container.insertBefore(draggedCard, target.nextSibling);
}
}
}

function touchEnd(event) {
if (!draggedCard) return;

const card = draggedCard;
const targetCategoryElement = card.closest('.card-container');
const targetCategory = targetCategoryElement ? targetCategoryElement.id : null;

if (isAdmin && card && targetCategory) {
updateCardCategory(card, targetCategory);
saveCardOrder();
}
cleanupDragState();
}

function findCardUnderTouch(x, y) {
const cards = document.querySelectorAll('.card:not(.dragging)');
return Array.from(cards).find(card => {
const rect = card.getBoundingClientRect();
return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
});
}

// PC端拖拽卡片
function dragStart(event) {
if (!isAdmin) {
event.preventDefault();
return;
}
draggedCard = event.target.closest('.card');
if (!draggedCard) return;

draggedCard.classList.add('dragging');
event.dataTransfer.effectAllowed = "move";
}

function dragOver(event) {
if (!isAdmin) {
event.preventDefault();
return;
}
event.preventDefault();
const target = event.target.closest('.card');
if (target && target !== draggedCard) {
const container = target.parentElement;
const mousePositionX = event.clientX;
const targetRect = target.getBoundingClientRect();

if (mousePositionX < targetRect.left + targetRect.width / 2) {
container.insertBefore(draggedCard, target);
} else {
container.insertBefore(draggedCard, target.nextSibling);
}
}
}

// 清理拖拽状态函数
function cleanupDragState() {
if (draggedCard) {
draggedCard.classList.remove('dragging');
draggedCard.style.transform = '';
draggedCard = null;
}

document.removeEventListener('touchmove', touchMove);
document.removeEventListener('touchend', touchEnd);

touchStartX = null;
touchStartY = null;
}

// PC端拖拽结束
function drop(event) {
if (!isAdmin) {
event.preventDefault();
return;
}
event.preventDefault();

const card = draggedCard;
const targetCategoryElement = event.target.closest('.card-container');
const targetCategory = targetCategoryElement ? targetCategoryElement.id : null;

validateToken().then(isValid => {
if (isValid && card && targetCategory) {
updateCardCategory(card, targetCategory);
saveCardOrder().catch(error => {
console.error('Save failed:', error);
});
}
cleanupDragState();
});
}

function dragEnd(event) {
if (draggedCard) {
draggedCard.classList.remove('dragging');
}
}

// 防抖变量
let saveCardOrderTimeoutId = null;
let pendingSaveCardOrder = false;

// 优化后的保存卡片顺序函数（带防抖）
async function saveCardOrder(debounceMs = 1000) {
console.log('DEBUG: saveCardOrder called.');

// 如果已有防抖定时器，清除它并重新设置
if (saveCardOrderTimeoutId) {
clearTimeout(saveCardOrderTimeoutId);
}

// 设置新的防抖定时器
saveCardOrderTimeoutId = setTimeout(async () => {
try {
if (!await validateToken()) {
console.log('DEBUG: saveCardOrder - Token validation failed, aborting save.');
return;
}

// 1. 从当前的全局categories对象创建一个深拷贝。
//    这个深拷贝将作为我们更新的基础，它包含了所有分类及其内部链接的当前状态和顺序。
//    这样，即使某些分类在DOM中是收起的，它们的数据也不会被丢失。
let updatedCategories = JSON.parse(JSON.stringify(categories));
console.log('DEBUG: saveCardOrder - Initial updatedCategories (deep copy):', JSON.parse(JSON.stringify(updatedCategories)));

// 2. 获取所有在DOM中渲染的 card-container 元素。
//    这些容器可能包含实际的卡片（如果分类是展开的），也可能为空（如果分类是收起的）。
const renderedContainers = document.querySelectorAll('.card-container');

// 3. 遍历这些容器，但只对那些“实际包含书签卡片”的分类进行数据更新。
renderedContainers.forEach(container => {
const categoryName = container.id; // 容器的ID就是分类名称
const cardElements = container.querySelectorAll('.card'); // 获取容器内所有的书签卡片元素

// 只有当这个容器内确实有书签卡片时，才认为这个分类是“展开且可能被修改”的。
if (cardElements.length > 0) {
// 确保 updatedCategories[categoryName] 存在且有 links 数组
if (!updatedCategories[categoryName]) {
updatedCategories[categoryName] = { isPrivate: false, links: [] };
console.warn('DEBUG: saveCardOrder - Created missing category in updatedCategories:', categoryName);
}
updatedCategories[categoryName].links = []; // 清空该分类的链接，准备按DOM顺序重新填充。
console.log('DEBUG: saveCardOrder - Clearing links for category:', categoryName, 'in updatedCategories.');

// 遍历DOM中的书签卡片元素，并根据其顺序重新构建该分类的链接列表。
cardElements.forEach(cardElement => {
const url = cardElement.getAttribute('data-url');
// 从全局links数组中找到完整的链接对象，以保留所有属性（如tips, icon, status等）。
const originalLink = links.find(link => link.url === url);

if (originalLink) {
const updatedLink = {
...originalLink,
category: categoryName,
isPrivate: cardElement.dataset.isPrivate === 'true'
};
updatedCategories[categoryName].links.push(updatedLink);
console.log('DEBUG: saveCardOrder - Added link:', updatedLink.name, 'to category:', categoryName, 'with isPrivate:', updatedLink.isPrivate);
} else {
console.warn('DEBUG: saveCardOrder - Original link not found for URL:', url);
}
});
} else {
console.log('DEBUG: saveCardOrder - Skipping empty or collapsed category:', categoryName);
}
});

// 4. 根据已经更新的 updatedCategories 对象，重新构建扁平化的 publicLinks, privateLinks 和 links 数组。
//    这确保了所有全局链接数组与结构化的 categories 对象保持一致。
publicLinks = [];
privateLinks = [];
links = []; // 主的扁平化链接列表

Object.keys(updatedCategories).forEach(catName => {
const categoryData = updatedCategories[catName];
// 确保 updatedCategories[catName] 是一个包含 links 数组的对象
if (categoryData && Array.isArray(categoryData.links)) {
categoryData.links.forEach(link => {
if (link.isPrivate) {
privateLinks.push(link);
} else {
publicLinks.push(link);
}
links.push(link); // 添加到主的扁平化列表
});
}
});
console.log('DEBUG: saveCardOrder - Rebuilt publicLinks (count:', publicLinks.length, '), privateLinks (count:', privateLinks.length, '), and links (count:', links.length, ').');

// 5. 将更新后的 updatedCategories 赋值回全局的 categories 对象。
categories = updatedCategories;
console.log('DEBUG: saveCardOrder - Global categories object updated to:', JSON.parse(JSON.stringify(categories)));

// 6. 将更新后的数据发送到后端进行持久化保存。
const response = await fetch('/api/saveOrder', {
method: 'POST',
headers: {
'Content-Type': 'application/json',
'Authorization': localStorage.getItem('authToken')
},
body: JSON.stringify({
userId: 'testUser',
links: links, // 发送完整的扁平化链接列表
categories: categories // 发送完整的结构化分类对象
}),
});
const result = await response.json();
if (!result.success) {
throw new Error('Failed to save order: ' + (result.message || 'Unknown error'));
}
console.log("DEBUG: Order saved successfully! Result:", result);
showToast('顺序已保存！');
} catch (error) {
console.error('ERROR: Error saving order:', error);
await customAlert('保存顺序失败，请重试', '保存失败');
}
console.log('DEBUG: saveCardOrder finished.');

// 重置防抖标志
pendingSaveCardOrder = false;
}, debounceMs);

// 设置防抖标志
pendingSaveCardOrder = true;
}

tooltip.textContent = tips;
}

tooltip.style.display = 'block';

const offsetX = 15;
const offsetY = 10;

const tooltipRect = tooltip.getBoundingClientRect();
const pageWidth = window.innerWidth;
const pageHeight = window.innerHeight;

let left = e.pageX + offsetX;
let top = e.pageY + offsetY;

if (pageWidth - e.clientX < 200) {
left = e.pageX - tooltipRect.width - offsetX;
}
// 如果距离底部小于100像素，往上显示
if (pageHeight - e.clientY < 100) {
top = e.pageY - tooltipRect.height - offsetY;
}

tooltip.style.left = left + 'px';
tooltip.style.top = top + 'px';
}

// 前端检查是否有 token
async function validateToken() {
console.log('DEBUG: validateToken called.');
const token = localStorage.getItem('authToken');
if (!token) {
isLoggedIn = false;
updateUIState();
console.log('DEBUG: validateToken - No authToken found, isLoggedIn set to false.');
return false;
}

try {
const response = await fetch('/api/getLinks?userId=testUser', {
headers: { 'Authorization': token }
});

if (response.status === 401) {
const errorData = await response.json(); // Attempt to parse error response
await resetToLoginState('token已过期，请重新登录。' + (errorData.message || ''));
console.log('DEBUG: validateToken - Token expired or invalid (401).');
return false;
}

isLoggedIn = true;
updateUIState();
console.log('DEBUG: validateToken - Token valid, isLoggedIn set to true.');
return true;
} catch (error) {
// 🔧 安全修复：避免泄露详细错误信息
console.error('ERROR: Token validation failed:', error); // DEBUG
// If fetch itself fails (e.g., network error), it's not a 401, so reset to login state
await resetToLoginState('验证登录状态失败，请重新登录。' + error.message);
return false;
}
}

if (!isValidHttpUrl(item.url)) {
console.warn('DEBUG: 导入时跳过无效URL协议的链接:', item.url);

console.warn('DEBUG: 导入时跳过重复链接 (URL已存在或本次导入中重复):', item.url);
continue; // 跳过此链接，因为它已经存在或在本次导入中重复

icon: item.logo || '',

    constantTimeCompare(a, b) {
        if (a.length !== b.length) return false;
        let result = 0;
        for (let i = 0; i < a.length; i++) {
            result |= a.charCodeAt(i) ^ b.charCodeAt(i);
        }
        return result === 0;

    // 服务端 token 验证
    async validateServerToken(authToken, env) {
        console.log('DEBUG: Backend validateServerToken called.');
        if (!authToken) {
            console.log('DEBUG: Backend validateServerToken - No authToken provided.');
            return {
                isValid: false,
                status: 401,
                response: { error: 'Unauthorized', message: '未登录或登录已过期' }
            };
        }

        try {
            const [timestamp, hash] = authToken.split('.');
            const tokenTimestamp = parseInt(timestamp);
            const now = Date.now();

            // Token 过期时间设置为 30 天
            const THIRTY_DAYS_IN_MS = 30 * 24 * 60 * 60 * 1000;
            if (now - tokenTimestamp > THIRTY_DAYS_IN_MS) {
                console.log('DEBUG: Backend validateServerToken - Token expired.');
                return {
                    isValid: false,
                    status: 401,
                    response: {
                        error: 'Token expired',
                        tokenExpired: true,
                        message: '登录已过期，请重新登录'
                    }
                };
            }

            // 检查 ADMIN_PASSWORD 是否已设置
            if (!env.ADMIN_PASSWORD) {
                console.error('ERROR: ADMIN_PASSWORD environment variable is not set in Cloudflare Worker.');
                return {
                    isValid: false,
                    status: 500, // Internal Server Error, as it's a configuration issue
                    response: {
                        error: 'Server configuration error',
                        message: '服务器配置错误：ADMIN_PASSWORD 未设置，请联系管理员。'
                    }
                };
            }

            const tokenData = timestamp + "_" + env.ADMIN_PASSWORD;
            const encoder = new TextEncoder();
            const data = encoder.encode(tokenData);
            const hashBuffer = await crypto.subtle.digest('SHA-256', data);
            const expectedHash = btoa(String.fromCharCode(...new Uint8Array(hashBuffer)));

            // 使用常量时间比较防止时序攻击
            if (!this.constantTimeCompare(hash, expectedHash)) { // 注意这里调用了 this.constantTimeCompare

            return { isValid: true, status: 200, response: { message: 'Token valid' } };
